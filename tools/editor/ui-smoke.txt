# The regression test of the editor's Qt side (editor/ui/UiScript.h has the commands):
#   tools\editor\run-ui-smoke.ps1
# builds a map from nothing through the menus, the toolbar and the panels, compiles and plays it,
# and cancels the build of a retail map. Every check is a PASS or FAIL line in ui-smoke.txt.log.

wait 2000
expect docState text No map
expect actionBuild enabled false

# File > New Map
trigger actionNewMap
set newMapDialog editor/uismoke
accept newMapDialog
waitfor docState text 15000 Uncompiled
expectcontains window windowTitle editor/uismoke
expect actionBuild enabled true
expect actionBuildCancel enabled false
expectitem outlinerTree 0 1 worldspawn
expect outlinerTree worldBrushes 0

# the box tool: a drag on the grid plane, seen from above, is one brush and one step of the history
set gridCombo 16
set boxHeightSpin 64
trigger actionToolBox
expect actionToolBox checked true
engine editorView 0 0 600 0 90
wait 300
engine editorDrag 0.5 0.5 0.6 0.62
wait 300
expect outlinerTree worldBrushes 1
expect orthoView numBrushes 1
expect orthoView numSelected 1
expectcontains window windowTitle editor/uismoke* [uncompiled]
expectitem faceList 0 1 textures/base_wall/lfwall13f3
trigger actionDeselect
expect actionToolSelect checked true
trigger actionUndo
wait 300
expect outlinerTree worldBrushes 0
expect actionRedo enabled true

# a room: a box made hollow, a light and a player start
engine editorCreateBrush -256 -128 0 0 128 128
wait 200
trigger actionHollow
accept hollowDialog
wait 300
expect outlinerTree worldBrushes 6
engine editorCreate light
engine editorSetKey light_1 origin "-128 0 96"
engine editorSetKey light_1 light_radius "320 320 320"
engine editorCreate info_player_start
engine editorSetKey info_player_start_1 origin "-136 0 24"
wait 300

# the gizmo of an entity, through the viewport's mouse path: seen from above the x arrow points up
engine editorSelect light_1
engine editorView -128 0 600 0 90
wait 300
engine editorDrag 0.5 0.44 0.5 0.3
waitfor actionUndo text 5000 &Undo Move*
trigger actionUndo
wait 200

# Build, Play, Stop
trigger actionBuild
waitfor docState text 120000 Compiled
expectcontains buildResult text Build OK: 1 areas, 0 portals
expectcontains outputView plainText Build OK: 1 areas, 0 portals
expectcontains window windowTitle editor/uismoke -
trigger actionPlay
waitfor docState text 60000 Playing
expect actionBuild enabled false
expect actionStop enabled true
wait 3000
trigger actionStop
waitfor docState text 60000 Compiled
wait 500

# a map of that name exists now: New Map asks, and No keeps the open one
trigger actionNewMap
set newMapDialog editor/uismoke
accept newMapDialog
expect newMapOverwrite visible true
reject newMapOverwrite
wait 300
expect outlinerTree worldBrushes 6

# a material and texture values through the Inspector, on one face
engine editorSelectBrush world 1
wait 300
expectitem faceList 2 1 textures/base_wall/lfwall13f3
select faceList 2
wait 200
set faceMaterialEdit textures/common/nodraw
wait 200
expectitem faceList 2 1 textures/common/nodraw
expectitem faceList 0 1 textures/base_wall/lfwall13f3
set faceShiftS 16
wait 200
expectitem faceList 2 2 16
expect docState text Uncompiled
# and from the Content Browser
set contentFilter textures/editor/visportal
wait 1500
select contentView 0
click contentApplyMaterial
wait 200
expectitem faceList 2 1 textures/editor/visportal
trigger actionUndo
trigger actionUndo
trigger actionUndo
wait 300
engine editorSelectBrush world 1
wait 300
expectitem faceList 2 1 textures/base_wall/lfwall13f3
expectitem faceList 2 2 0

# the 2D view: a click selects, a drag moves the selection on the grid, the box tool draws, the middle button pans
engine editorView -128 0 300 0 90
wait 300
click orthoCenter
trigger actionDeselect
wait 300
expect orthoView numBrushes 6
expect orthoView numSelected 0
drag orthoView 0.5+30 0.5+30 0.5+30 0.5+30
wait 300
expect orthoView numSelected 1
drag orthoView 0.5+30 0.5+30 0.5+62 0.5+30
wait 300
expect actionUndo text &Undo Move brush
trigger actionUndo
wait 300
trigger actionToolBox
drag orthoView 0.5+200 0.5+30 0.5+232 0.5+62
wait 300
expect outlinerTree worldBrushes 7
expect actionUndo text &Undo Create brush
trigger actionUndo
trigger actionToolSelect
drag orthoView 0.5 0.5 0.5+40 0.5 middle
wait 300
expect orthoView numBrushes 6
engine editorView -128 0 64 0 0
wait 300
set orthoKind Front (XZ)
wait 300
expect orthoView numBrushes 6
set orthoKind Top (XY)

# Ctrl+D on a brush: a copy, selected, one step
engine editorSelectBrush world 1
trigger actionDuplicate
wait 300
expect outlinerTree worldBrushes 7
expect actionUndo text &Undo Duplicate brush
expect orthoView numSelected 1
trigger actionUndo
wait 300
expect outlinerTree worldBrushes 6

# the rotate gizmo on a brush, through the viewport's mouse path: seen from above the z ring is a
# circle, from its right end to straight up is a quarter turn
engine editorCreateBrush -192 -16 32 -64 16 48
set angleCombo 45
trigger actionGizmoRotate
engine editorView -128 0 600 0 90
wait 300
engine editorDrag 0.575 0.5 0.5 0.3
waitfor actionUndo text 5000 &Undo Rotate brush
engine editorBrushInfo
wait 300
expectcontains outputView plainText bounds (-144 -64 32) (-112 64 48)
trigger actionUndo
wait 200
engine editorSelectBrush world 6
engine editorRotateBrush x 90
wait 200
expect actionUndo text &Undo Rotate brush
trigger actionUndo
wait 200
engine editorSelectBrush world 6
trigger actionDelete
trigger actionGizmoMove
set angleCombo 15
wait 300
expect outlinerTree worldBrushes 6

# the grid of the 3D viewport
expect actionShowGrid checked true
trigger actionShowGrid
expect actionShowGrid checked false
trigger actionShowGrid
expect actionShowGrid checked true

# Play on a map whose brushes changed since the build: it builds first, then plays
expect docState text Uncompiled
expect buildResult builds 1
trigger actionPlay
waitfor docState text 120000 Playing
expect buildResult builds 2
expectcontains buildResult text Build OK: 1 areas, 0 portals
wait 2000
trigger actionStop
waitfor docState text 60000 Compiled
wait 500

# a leak: without the ceiling the build says so and the leak can be shown
expect actionGoToLeak enabled false
engine editorSelectBrush world 4
trigger actionDelete
wait 300
expect outlinerTree worldBrushes 5
trigger actionBuild
waitfor buildResult text 120000 Build LEAKED*
wait 500
accept buildMessage
wait 300
expect docState text Uncompiled
expect actionGoToLeak enabled true
expect actionShowLeak checked true
trigger actionGoToLeak
trigger actionUndo
wait 300
expect outlinerTree worldBrushes 6

# a second room, a doorway with a visportal, a low wall and a monster behind it
engine editorCreateBrush 0 -128 0 256 128 128
wait 200
trigger actionHollow
accept hollowDialog
wait 200
engine editorSelectBrush world 7
trigger actionDelete
engine editorSelectBrush world 0
trigger actionDelete
engine editorCreateBrush -16 -128 0 16 -32 128
engine editorCreateBrush -16 32 0 16 128 128
engine editorCreateBrush -16 -32 104 16 32 128
engine editorCreateBrush -8 -32 16 8 32 104 textures/editor/visportal
engine editorCreateBrush 120 -112 16 136 24 56
engine editorCreate light
engine editorSetKey light_2 origin "128 0 96"
engine editorSetKey light_2 light_radius "320 320 320"
engine editorCreate monster_zombie_fat
engine editorSetKey monster_zombie_fat_1 origin "200 -40 24"
wait 300
expect outlinerTree worldBrushes 15
trigger actionBuild
waitfor docState text 180000 Compiled
expectcontains buildResult text Build OK: 2 areas, 1 portals
expectcontains buildResult text 1 AAS files
# Play builds nothing now and Stop comes back to the compiled map
trigger actionPlay
waitfor docState text 60000 Playing
wait 3000
trigger actionStop
waitfor docState text 60000 Compiled

# Make Entity and Move To World
engine editorSelectBrush world 14
trigger actionMakeEntity
set makeEntityDialog func_static
accept makeEntityDialog
wait 300
expect outlinerTree worldBrushes 14
trigger actionMoveToWorld
wait 300
expect outlinerTree worldBrushes 15

# a patch is selected and edited as a whole: click, gizmo, 2D view, Ctrl+D, Delete, material, turn, build
engine editorOpenMap editor/patch1
waitfor window windowTitle 30000 editor/patch1*
wait 500
expect outlinerTree worldBrushes 7
engine editorView 0 0 76 0 90
wait 300
engine editorClick 0.5 0.5
wait 300
expect faceList isPatch true
expectitem faceList 0 0 patch
expect faceShiftS enabled false
# seen from above the y arrow points to the left of the screen
engine editorDrag 0.4625 0.5 0.25 0.5
waitfor actionUndo text 5000 &Undo Move patch
trigger actionUndo
wait 200
click orthoCenter
trigger actionDeselect
wait 300
expect orthoView numSelected 0
drag orthoView 0.5-12 0.5-12 0.5-12 0.5-12
wait 300
expect orthoView numSelected 1
expect faceList isPatch true
drag orthoView 0.5-12 0.5-12 0.5+20 0.5-12
wait 300
expect actionUndo text &Undo Move patch
trigger actionUndo
wait 200
engine editorSelectBrush world 6
trigger actionDuplicate
wait 300
expect outlinerTree worldBrushes 8
expect actionUndo text &Undo Duplicate patch
expect faceList isPatch true
trigger actionDelete
wait 300
expect outlinerTree worldBrushes 7
engine editorSelectBrush world 6
engine editorRotateBrush z 45
wait 200
expect actionUndo text &Undo Rotate patch
trigger actionUndo
wait 200
engine editorSelectBrush world 6
wait 200
set faceMaterialEdit _default
wait 300
expectitem faceList 0 1 _default
expect actionUndo text &Undo Set material of patch
# compiled: the patch is in the world the game loads, and still picked
trigger actionBuild
waitfor docState text 120000 Compiled
expectcontains buildResult text Build OK: 1 areas, 0 portals
trigger actionDeselect
engine editorView 0 0 76 0 90
wait 300
engine editorClick 0.5 0.5
wait 300
expect faceList isPatch true
expect orthoView numSelected 1

# Cancel: the compile of a retail map takes a minute, the Build menu stays alive in it
engine editorOpenMap game/mc_underground
waitfor window windowTitle 120000 game/mc_underground*
wait 1000
trigger actionBuild
waitfor docState text 30000 Building
wait 4000
expect actionBuildCancel enabled true
expect actionPlay enabled false
expect buildProgress visible true
trigger actionBuildCancel
waitfor buildResult text 60000 Build cancelled
wait 500
expect docState text Compiled
expect actionBuild enabled true
# the editor is usable after it
engine editorSelect info_player_start_1
wait 500
expect orthoView numSelected 1
trigger actionCloseMap
waitfor docState text 30000 No map
quit
